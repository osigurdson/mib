cmake_minimum_required(VERSION 3.20)
project(tiny_vi LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
    add_compile_options(/W4 /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(OUTPUT "${GENERATED}/font-data.h"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed-font.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/fonts/editor.tvf" "${GENERATED}/font-data.h"
    DEPENDS tools/embed-font.py fonts/editor.tvf VERBATIM)
add_custom_target(font-data DEPENDS "${GENERATED}/font-data.h")

add_library(tinyvi-core STATIC src/app.c src/editor.c src/render.c src/tiny-font.c)
add_dependencies(tinyvi-core font-data)
target_include_directories(tinyvi-core PUBLIC src PRIVATE "${GENERATED}")

if(WIN32)
    set(PLATFORM_MEMORY src/platform/windows/memory.c)
    set(PLATFORM_FILES src/platform/windows/files_stub.c)
    set(PLATFORM_WINDOW src/platform/windows/window_stub.c)
    message(STATUS "Windows backend: memory implemented; window and file operations are stubs")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(PLATFORM_MEMORY src/platform/linux/memory.c)
    set(PLATFORM_FILES src/platform/linux/files.c)
    set(PLATFORM_WINDOW src/platform/linux/wayland.c)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client xkbcommon)
    pkg_get_variable(PROTOCOLS_DIR wayland-protocols pkgdatadir)
    find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
    if(NOT PROTOCOLS_DIR)
        message(FATAL_ERROR "wayland-protocols development package is required")
    endif()
    set(PROTOCOL_NAMES xdg-shell viewporter fractional-scale-v1)
    set(PROTOCOL_XMLS
        "${PROTOCOLS_DIR}/stable/xdg-shell/xdg-shell.xml"
        "${PROTOCOLS_DIR}/stable/viewporter/viewporter.xml"
        "${PROTOCOLS_DIR}/staging/fractional-scale/fractional-scale-v1.xml")
    foreach(NAME XML IN ZIP_LISTS PROTOCOL_NAMES PROTOCOL_XMLS)
        set(HEADER "${GENERATED}/${NAME}-client-protocol.h")
        set(SOURCE "${GENERATED}/${NAME}-protocol.c")
        add_custom_command(OUTPUT "${HEADER}" "${SOURCE}"
            COMMAND "${WAYLAND_SCANNER}" client-header "${XML}" "${HEADER}"
            COMMAND "${WAYLAND_SCANNER}" private-code "${XML}" "${SOURCE}"
            DEPENDS "${XML}" VERBATIM)
        list(APPEND PROTOCOL_OUTPUTS "${HEADER}" "${SOURCE}")
    endforeach()
    add_library(tinyvi-protocols STATIC ${PROTOCOL_OUTPUTS})
    target_link_libraries(tinyvi-protocols PUBLIC PkgConfig::WAYLAND)
    target_include_directories(tinyvi-protocols PUBLIC "${GENERATED}")
else()
    message(FATAL_ERROR "Supported targets are Linux and the Windows stub backend")
endif()

add_library(tinyvi-memory STATIC ${PLATFORM_MEMORY})
target_include_directories(tinyvi-memory PUBLIC src)
add_executable(tiny-vi src/main.c ${PLATFORM_WINDOW} ${PLATFORM_FILES})
target_link_libraries(tiny-vi PRIVATE tinyvi-core tinyvi-memory)
if(TARGET tinyvi-protocols)
    target_link_libraries(tiny-vi PRIVATE tinyvi-protocols)
endif()

include(CTest)
if(BUILD_TESTING)
    # Assertions perform test actions too; keep them enabled in every build type.
    if(MSVC)
        add_compile_options(/UNDEBUG)
    else()
        add_compile_options(-UNDEBUG)
    endif()
    add_executable(font-test tests/font-test.c)
    add_dependencies(font-test font-data)
    target_include_directories(font-test PRIVATE "${GENERATED}")
    target_link_libraries(font-test PRIVATE tinyvi-memory)
    add_test(NAME font COMMAND font-test)

    add_executable(app-test tests/app-test.c ${PLATFORM_FILES})
    target_link_libraries(app-test PRIVATE tinyvi-core tinyvi-memory)
    add_test(NAME app COMMAND app-test)

    # This test exercises the explicit failure contract on either host OS.
    add_executable(platform-stub-test tests/platform-stub-test.c
        src/platform/windows/files_stub.c src/platform/windows/window_stub.c)
    target_link_libraries(platform-stub-test PRIVATE tinyvi-core tinyvi-memory)
    add_test(NAME platform-stub COMMAND platform-stub-test)

    if(TARGET tinyvi-protocols)
        add_executable(editor-test tests/editor-test.c ${PLATFORM_FILES})
        target_link_libraries(editor-test PRIVATE tinyvi-core tinyvi-memory)
        add_test(NAME editor COMMAND editor-test)
        add_executable(client-test tests/client-test.c ${PLATFORM_FILES})
        target_link_libraries(client-test PRIVATE tinyvi-core tinyvi-memory tinyvi-protocols)
        add_test(NAME wayland-input COMMAND client-test)
    endif()
endif()
