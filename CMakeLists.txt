cmake_minimum_required(VERSION 3.20)
project(mib LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
    add_compile_options(/W4 /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
    add_link_options(-Wl,--gc-sections)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(OUTPUT "${GENERATED}/font-data.h"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed-font.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/fonts/editor.tvf" "${GENERATED}/font-data.h"
    DEPENDS tools/embed-font.py fonts/editor.tvf VERBATIM)
add_custom_target(font-data DEPENDS "${GENERATED}/font-data.h")

add_library(mib-core STATIC src/app.c src/session.c src/pages.c src/media.c src/image.c src/editor.c src/keys.c src/ex.c src/history.c src/substitute.c src/markdown.c src/render.c src/navigator.c src/help.c src/tiny-font.c)
add_dependencies(mib-core font-data)
target_include_directories(mib-core PUBLIC src PRIVATE "${GENERATED}")

if(WIN32)
    set(PLATFORM_MEMORY src/platform/windows/memory.c)
    set(PLATFORM_FILES src/platform/windows/files.c src/platform/windows/utf.c)
    set(PLATFORM_WINDOW src/platform/windows/window.c)
    set(PLATFORM_ENTRY src/platform/windows/entry.c)
    set(PLATFORM_LIBRARIES user32 gdi32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(PLATFORM_ENTRY src/platform/linux/entry.c)
    set(PLATFORM_MEMORY src/platform/linux/memory.c)
    set(PLATFORM_FILES src/platform/linux/files.c)
    set(PLATFORM_WINDOW src/platform/linux/wayland.c)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client xkbcommon)
    pkg_get_variable(PROTOCOLS_DIR wayland-protocols pkgdatadir)
    find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
    if(NOT PROTOCOLS_DIR)
        message(FATAL_ERROR "wayland-protocols development package is required")
    endif()
    set(PROTOCOL_NAMES xdg-shell viewporter fractional-scale-v1
                       xdg-decoration-unstable-v1 cursor-shape-v1 tablet-unstable-v2)
    set(PROTOCOL_XMLS
        "${PROTOCOLS_DIR}/stable/xdg-shell/xdg-shell.xml"
        "${PROTOCOLS_DIR}/stable/viewporter/viewporter.xml"
        "${PROTOCOLS_DIR}/staging/fractional-scale/fractional-scale-v1.xml"
        "${PROTOCOLS_DIR}/unstable/xdg-decoration/xdg-decoration-unstable-v1.xml"
        "${PROTOCOLS_DIR}/staging/cursor-shape/cursor-shape-v1.xml"
        "${PROTOCOLS_DIR}/unstable/tablet/tablet-unstable-v2.xml")
    foreach(NAME XML IN ZIP_LISTS PROTOCOL_NAMES PROTOCOL_XMLS)
        set(HEADER "${GENERATED}/${NAME}-client-protocol.h")
        set(SOURCE "${GENERATED}/${NAME}-protocol.c")
        add_custom_command(OUTPUT "${HEADER}" "${SOURCE}"
            COMMAND "${WAYLAND_SCANNER}" client-header "${XML}" "${HEADER}"
            COMMAND "${WAYLAND_SCANNER}" private-code "${XML}" "${SOURCE}"
            DEPENDS "${XML}" VERBATIM)
        list(APPEND PROTOCOL_OUTPUTS "${HEADER}" "${SOURCE}")
    endforeach()
    add_library(mib-protocols STATIC ${PROTOCOL_OUTPUTS})
    target_link_libraries(mib-protocols PUBLIC PkgConfig::WAYLAND)
    target_include_directories(mib-protocols PUBLIC "${GENERATED}")
else()
    message(FATAL_ERROR "Supported targets are Linux and Windows")
endif()

add_library(mib-memory STATIC ${PLATFORM_MEMORY})
target_include_directories(mib-memory PUBLIC src)
# mib.exe is a GUI program, so starting it opens no console window, but it keeps the Unicode
# wmain entry point. WIN32 sets the GUI subsystem for MSVC and GCC; Zig's Clang needs it as a
# linker flag.
add_executable(mib WIN32 src/main.c ${PLATFORM_ENTRY} ${PLATFORM_WINDOW} ${PLATFORM_FILES})
target_link_libraries(mib PRIVATE mib-core mib-memory ${PLATFORM_LIBRARIES})
# version.h names the git version and commit; checked on every build, rewritten on change.
add_custom_target(version
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/version.py"
            "${GENERATED}/version.h"
    BYPRODUCTS "${GENERATED}/version.h" VERBATIM)
add_dependencies(mib version)
target_include_directories(mib PRIVATE "${GENERATED}")
if(MSVC)
    target_link_options(mib PRIVATE /ENTRY:wmainCRTStartup)
elseif(WIN32)
    target_link_options(mib PRIVATE -municode -Wl,--subsystem,windows)
endif()
if(TARGET mib-protocols)
    target_link_libraries(mib PRIVATE mib-protocols)
endif()

include(CTest)
if(BUILD_TESTING)
    # Assertions perform test actions too; keep them enabled in every build type.
    if(MSVC)
        add_compile_options(/UNDEBUG)
    else()
        add_compile_options(-UNDEBUG)
    endif()
    add_executable(font-test tests/font-test.c)
    add_dependencies(font-test font-data)
    target_include_directories(font-test PRIVATE "${GENERATED}")
    target_link_libraries(font-test PRIVATE mib-memory)
    add_test(NAME font COMMAND font-test)

    add_executable(media-test tests/media-test.c ${PLATFORM_FILES})
    target_link_libraries(media-test PRIVATE mib-core mib-memory ${PLATFORM_LIBRARIES})
    add_test(NAME media COMMAND media-test)

    add_executable(session-test tests/session-test.c ${PLATFORM_FILES})
    target_link_libraries(session-test PRIVATE mib-core mib-memory ${PLATFORM_LIBRARIES})
    add_test(NAME session COMMAND session-test)

    add_executable(modal-test tests/modal-test.c ${PLATFORM_FILES})
    target_link_libraries(modal-test PRIVATE mib-core mib-memory)
    add_test(NAME modal COMMAND modal-test)

    add_executable(app-test tests/app-test.c ${PLATFORM_FILES})
    target_link_libraries(app-test PRIVATE mib-core mib-memory)
    add_test(NAME app COMMAND app-test)

    if(WIN32)
        add_executable(windows-files-test tests/windows-files-test.c ${PLATFORM_FILES})
        target_link_libraries(windows-files-test PRIVATE mib-core mib-memory)
        add_test(NAME windows-files COMMAND windows-files-test)
        add_executable(windows-clipboard-test tests/windows-clipboard-test.c ${PLATFORM_FILES})
        target_link_libraries(windows-clipboard-test PRIVATE mib-core mib-memory user32 gdi32)
        add_test(NAME windows-clipboard COMMAND windows-clipboard-test)
        add_executable(windows-window-test tests/windows-window-test.c ${PLATFORM_FILES})
        # windows-window-test includes the window callbacks directly for message coverage.
        target_link_libraries(windows-window-test PRIVATE mib-core mib-memory user32 gdi32)
        add_test(NAME windows-window COMMAND windows-window-test)
        add_executable(windows-e2e-test tests/windows-e2e-test.c src/platform/windows/utf.c)
        target_include_directories(windows-e2e-test PRIVATE src)
        target_link_libraries(windows-e2e-test PRIVATE user32)
        add_dependencies(windows-e2e-test mib)
        add_test(NAME windows-e2e COMMAND windows-e2e-test)
        set_tests_properties(windows-files windows-window windows-e2e PROPERTIES TIMEOUT 45)
    endif()
    if(TARGET mib-protocols)
        find_program(TESSERACT tesseract)
        if(TESSERACT)
            add_executable(ocr-test tests/ocr-test.c ${PLATFORM_FILES})
            target_link_libraries(ocr-test PRIVATE mib-core mib-memory)
            add_test(NAME ocr COMMAND ocr-test)
            set_tests_properties(ocr PROPERTIES TIMEOUT 20)
        endif()
        option(MIB_LINUX_E2E "Run real-app workflows in a private Mutter compositor" OFF)
        if(MIB_LINUX_E2E)
            find_program(MUTTER mutter REQUIRED)
            find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
            add_test(NAME linux-e2e COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-linux-e2e.py"
                --binary "$<TARGET_FILE:mib>" --output "${CMAKE_CURRENT_BINARY_DIR}/e2e")
            set_tests_properties(linux-e2e PROPERTIES TIMEOUT 120)
        endif()
        pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
        add_executable(clipboard-test tests/clipboard-test.c ${PLATFORM_FILES})
        target_link_libraries(clipboard-test PRIVATE mib-core mib-memory mib-protocols PkgConfig::WAYLAND_SERVER)
        add_test(NAME wayland-clipboard COMMAND clipboard-test)
        set_tests_properties(wayland-clipboard PROPERTIES TIMEOUT 15)
        add_executable(editor-test tests/editor-test.c ${PLATFORM_FILES})
        target_link_libraries(editor-test PRIVATE mib-core mib-memory)
        add_test(NAME editor COMMAND editor-test)
        add_executable(client-test tests/client-test.c ${PLATFORM_FILES})
        target_link_libraries(client-test PRIVATE mib-core mib-memory mib-protocols PkgConfig::WAYLAND_SERVER)
        add_test(NAME wayland-input COMMAND client-test)
    endif()
endif()
